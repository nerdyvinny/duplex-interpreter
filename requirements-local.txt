# duplex-interpreter — fully local / offline extras
# Install on top of requirements.txt:
#   pip install -r requirements.txt -r requirements-local.txt
#
# These pull multi-GB models on first run. See README "Running fully offline".

# --- local STT ---
faster-whisper>=1.1.0     # CTranslate2 Whisper. GPU needs a CUDA-matched ctranslate2;
                          # CPU int8 with model_size=small is still near real-time.

# --- local MT ---
argostranslate>=1.9       # per-pair models ~100 MB, CPU only

# --- local TTS ---

# --- local stack extras, see README "Running fully offline" ---
# Windows GPU: CTranslate2 needs the CUDA math libraries; without these it
# fails with "Library cublas64_12.dll is not found" on a perfectly good card.
nvidia-cublas-cu12; platform_system == "Windows"
nvidia-cudnn-cu12; platform_system == "Windows"
